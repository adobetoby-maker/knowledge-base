# Overnight Batch: SEO Article Generation

## Overview

Generating a complete SEO article cluster (10-50 articles) overnight using Claude Haiku. This runs when the developer is not present — all decisions must be pre-specified.

## Pre-Batch Planning (Do This Before Starting)

Specify in a JSON config file before running:
```json
{
  "project": "jrs-auto-repair",
  "cluster": "maintenance-guides",
  "targetAudience": "Twin Falls ID car owners",
  "voiceAndTone": "Friendly, professional, local. Like a trusted mechanic neighbor.",
  "brandName": "Jr.'s Auto Repair",
  "location": "Twin Falls, ID and Magic Valley",
  "phone": "(208) 595-2101",
  "articles": [
    {
      "slug": "how-often-change-oil-synthetic",
      "title": "How Often Should You Change Synthetic Oil in Twin Falls?",
      "targetKeyword": "how often change synthetic oil Twin Falls",
      "secondaryKeywords": ["synthetic oil change interval", "full synthetic oil Twin Falls ID"],
      "wordCount": 800,
      "intent": "informational",
      "cta": "schedule oil change"
    }
  ]
}
```

## Batch Script Structure

```typescript
// scripts/generate-articles.ts
import Anthropic from '@anthropic-ai/sdk'
import fs from 'fs'

const client = new Anthropic()
const CONFIG_FILE = 'article-batch-config.json'
const PROGRESS_FILE = 'article-batch-progress.json'
const OUTPUT_FILE = 'lib/generated-articles.ts'

interface Progress {
  completed: string[]
  failed: string[]
  lastRunAt: string
}

async function generateArticle(config: ArticleConfig, projectConfig: ProjectConfig): Promise<string> {
  const response = await client.messages.create({
    model: 'claude-haiku-4-5',
    max_tokens: 2000,
    messages: [{
      role: 'user',
      content: `Write an SEO article for a local auto repair shop.

Brand: ${projectConfig.brandName}
Location: ${projectConfig.location}
Phone: ${projectConfig.phone}
Voice: ${projectConfig.voiceAndTone}

Article:
Title: ${config.title}
Target keyword: "${config.targetKeyword}"
Secondary keywords: ${config.secondaryKeywords.join(', ')}
Word count: ~${config.wordCount} words
User intent: ${config.intent}
CTA: ${config.cta}

Requirements:
- Use the target keyword in the H1, first paragraph, and a subheading
- Include secondary keywords naturally (once or twice each)
- Include the location (Twin Falls, Magic Valley) where natural
- End with a clear CTA to call ${projectConfig.phone} or book online
- No keyword stuffing — it should read naturally
- Format: just the article body as plain text, no JSON wrapper

Write the article body now:`
    }],
  })

  return response.content[0].type === 'text' ? response.content[0].text : ''
}

async function main() {
  const config = JSON.parse(fs.readFileSync(CONFIG_FILE, 'utf-8'))
  
  // Load or initialize progress
  let progress: Progress = { completed: [], failed: [], lastRunAt: '' }
  if (fs.existsSync(PROGRESS_FILE)) {
    progress = JSON.parse(fs.readFileSync(PROGRESS_FILE, 'utf-8'))
    console.log(`Resuming: ${progress.completed.length} already done, ${progress.failed.length} failed`)
  }

  const results: Array<{ slug: string; title: string; body: string }> = []
  
  for (const article of config.articles) {
    if (progress.completed.includes(article.slug)) {
      console.log(`Skipping ${article.slug} (already done)`)
      continue
    }

    try {
      console.log(`Generating: ${article.title}`)
      const body = await generateArticle(article, config)
      
      results.push({
        slug: article.slug,
        title: article.title,
        body,
      })
      
      progress.completed.push(article.slug)
      progress.lastRunAt = new Date().toISOString()
      fs.writeFileSync(PROGRESS_FILE, JSON.stringify(progress, null, 2))
      
      console.log(`  Done: ${body.length} chars`)
      
      // Rate limit: 1 second between requests
      await new Promise(r => setTimeout(r, 1000))
      
    } catch (error) {
      console.error(`  FAILED: ${article.slug}`, error)
      progress.failed.push(article.slug)
      fs.writeFileSync(PROGRESS_FILE, JSON.stringify(progress, null, 2))
    }
  }

  // Write output
  if (results.length > 0) {
    const output = `// Auto-generated by generate-articles.ts on ${new Date().toISOString()}
// Review before committing — check for accuracy, brand voice, factual claims

export const generatedArticles = ${JSON.stringify(results, null, 2)}
`
    fs.writeFileSync(OUTPUT_FILE, output)
    console.log(`\nWrote ${results.length} articles to ${OUTPUT_FILE}`)
  }

  // Summary
  console.log(`\nBatch complete:`)
  console.log(`  Completed: ${progress.completed.length}`)
  console.log(`  Failed: ${progress.failed.length}`)
  if (progress.failed.length > 0) {
    console.log(`  Failed slugs: ${progress.failed.join(', ')}`)
  }
}

main().catch(console.error)
```

## Running the Batch

```bash
# Start it (runs overnight)
npx tsx scripts/generate-articles.ts > logs/article-gen.log 2>&1 &

# Check progress in the morning
cat article-batch-progress.json
tail -50 logs/article-gen.log
```

## Post-Generation Human Review Checklist

Before adding generated articles to production:
- [ ] Read each article — check for factual errors about services
- [ ] Verify all CTAs use the correct phone number
- [ ] Check that location mentions are accurate (Twin Falls, not wrong city)
- [ ] Confirm no keyword stuffing or unnatural phrasing
- [ ] Verify the article answers the stated user intent
- [ ] Check the slug and title match the content

## Merging Into lib/articles.ts

After review, manually move approved articles from `generatedArticles` array into `lib/articles.ts`:
```typescript
// Add to lib/articles.ts after review
import { generatedArticles } from './generated-articles'

export const articles: Article[] = [
  ...existingArticles,
  ...generatedArticles.map(a => ({
    ...a,
    category: 'maintenance',
    date: '2026-05-18',
    readTime: estimateReadTime(a.body),
    excerpt: a.body.split('.').slice(0, 2).join('.') + '.',
  })),
]
```
